cmake_minimum_required(VERSION 3.25)
project(PickpocketOverhaul VERSION 1.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(ENABLE_SKYRIM_SE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(COMMONLIB_ENABLE_IPO OFF CACHE BOOL "" FORCE)
set(SKSE_SUPPORT_PATCH_SAFETY OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/CommonLibSSE-NG)
include(vendor/CommonLibSSE-NG/cmake/CommonLibSSE.cmake)
add_commonlibsse_plugin(PickpocketOverhaul NAME "Pickpocket Chance Rework" AUTHOR "Maxmc" VERSION 1.1.0 SOURCES src/Plugin.cpp src/QuickPocketIntegration.cpp)
target_include_directories(PickpocketOverhaul PRIVATE src)
target_precompile_headers(PickpocketOverhaul PRIVATE src/PCH.h)
target_compile_definitions(PickpocketOverhaul PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(PickpocketOverhaul PRIVATE /W4 /permissive- /Z7)
target_link_options(PickpocketOverhaul PRIVATE /DEBUG /OPT:REF /OPT:ICF)
set_target_properties(PickpocketOverhaul PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/package/SKSE/Plugins")
enable_testing()
add_executable(OverhaulTests tests/CoreTests.cpp)
target_include_directories(OverhaulTests PRIVATE src)
add_test(NAME OverhaulBehavior COMMAND OverhaulTests)

add_executable(QuickPocketHookTests tests/QuickPocketHookTests.cpp)
target_include_directories(QuickPocketHookTests PRIVATE src)
target_compile_definitions(QuickPocketHookTests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
add_test(NAME QuickPocketNativeHooks COMMAND QuickPocketHookTests)
add_executable(QuickPocketBridgeTests tests/QuickPocketBridgeTests.cpp)
target_include_directories(QuickPocketBridgeTests PRIVATE src)
add_test(NAME QuickPocketBridgeRouting COMMAND QuickPocketBridgeTests)
